# gate.sh: [ -f "$f.license" ] && head -5 "$f.license" | grep -q SPDX-License-Identifier
# The population is git ls-files; the COVERAGE is [ -f ]. An untracked sidecar covers a
# tracked file on this disk and does not exist in a fresh clone or in CI.

$ printf 'SPDX-...\n' > probe-sidecar.yaml.license      # deliberately NOT git added
$ git ls-files | grep -c '^probe-sidecar.yaml.license$'  ->  0
$ git ls-files | grep -c '^probe-sidecar.yaml$'          ->  1
